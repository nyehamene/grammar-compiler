Source = { declaration };

declaration = binding | package_directive | Rule | comment;
binding     = ident "=" directive ";";
package_directive = "@package" "(" string ")";
directive   = "@import" "(" string ")" | "@package" "(" string ")";
Rule        = ident "=" production ";";
comment     = "/" "/" /[^\n]*/ [ "\n" ];

production  = sequence | alternative;
alternative = sequence { "|" production };
sequence    = basic { basic };
basic       = group
            | repetition
            | optional
            | term
            ;

// basic, terminal, non_terminal and term should not
// appear in the syntax tree.
group          = "(" production ")";
repetition     = "{" production "}";
optional       = "[" production "]";
term           = terminal | non_teminal;
non_terminal   = ident | member_access;
terminal       = string | regexp | external_value;
external_value = "$" ident;

// member_access is left-associative
member_access    = ident "." ident { "." ident };
import_directive = "@import" "(" string ")";

string = /"[^"\n]*"/;
ident  = /[a-zA-Z_][a-zA-Z0-9_]*/;
regexp = "/" /[^\/]*/ "/";
